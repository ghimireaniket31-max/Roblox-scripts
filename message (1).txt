local TableInstanceMethods = {
    GetChildren = game.GetChildren,
    GetDescendants = game.GetDescendants,

local function IsProtected(Object)
    if not Object or typeof(Object) ~= "Instance" then return false end
    if ProtectedInstances[Object] then return true end
    local current = Object
    local depth = 0
    while depth < 10 do
        local success, parent = pcall(function() return current.Parent end)
        if not success or not parent or parent == game then break end
        if ProtectedInstances[parent] then return true end
        current = parent
        depth = depth + 1
    end
    return false
end

local function HookMetaMethod(Object, MetaMethod, Function)
    local MT = GetRawMT(Object)
    local Old = MT[MetaMethod]
    if hookmetamethod then
        return hookmetamethod(Object, MetaMethod, Function)
    else
        return HookFunction(Old, Function)
    end
end

for MethodName, MethodFunction in next, TableInstanceMethods do
    TblDataCache[MethodName] = HookFunction(MethodFunction, function(self, ...)
        if not CheckCaller() then
            local success, ReturnedTable = pcall(TblDataCache[MethodName], self, ...)
            if success and typeof(ReturnedTable) == "table" then
                for i = #ReturnedTable, 1, -1 do
                    if IsProtected(ReturnedTable[i]) then
                        table.remove(ReturnedTable, i)
                    end
                end
                return ReturnedTable
            end
        end
        return TblDataCache[MethodName](self, ...)
    end)
end

for MethodName, MethodFunction in next, FindInstanceMethods do
    FindDataCache[MethodName] = HookFunction(MethodFunction, function(self, ...)
        if not CheckCaller() then
            local success, FindResult = pcall(FindDataCache[MethodName], self, ...)
            if success and IsProtected(FindResult) then
                return nil
            end
        end
        return FindDataCache[MethodName](self, ...)
    end)
end

OriginalNameCall = HookMetaMethod(game, "__namecall", function(...)
    local self = (...)
    local NCMethod = GetNCMethod()
    
    if not CheckCaller() and OriginalNameCall then
        if typeof(self) == "Instance" then
            if TableInstanceMethods[NCMethod] then
                local success, ReturnedData = pcall(OriginalNameCall, ...)
                if success and typeof(ReturnedData) == "table" then
                    for i = #ReturnedData, 1, -1 do
                        if IsProtected(ReturnedData[i]) then
                            table.remove(ReturnedData, i)
                        end
                    end
                end
                return ReturnedData
            elseif FindInstanceMethods[NCMethod] then
                local success, FindResult = pcall(OriginalNameCall, ...)
                if success and IsProtected(FindResult) then
                    return nil
                end
                return FindResult
            end
        end
    end
    if OriginalNameCall then
        return OriginalNameCall(...)
    end
end)

OriginalNewIndex = HookMetaMethod(game, "__newindex", function(...)
    local self, Property, Value = ...
    if CheckCaller() and typeof(self) == "Instance" then
        if IsProtected(self) and not PropertyChangedData[self] and rawequal(Property, "Parent") then
            PropertyChangedData[self] = true
            pcall(function()
                self.Parent = Value
            end)
            PropertyChangedData[self] = false
            return
        end
    end
    if OriginalNewIndex then
        return OriginalNewIndex(...)
    end
end)

getgenv().ProtectInstance = function(NewInstance)
    if NewInstance and typeof(NewInstance) == "Instance" then
        ProtectedInstances[NewInstance] = true
        pcall(function()
            if syn and syn.protect_gui then
                syn.protect_gui(NewInstance)
            elseif gethui then
                NewInstance.Parent = gethui()
            end
        end)
    end
end

getgenv().UnProtectInstance = function(NewInstance)
    if NewInstance then
        ProtectedInstances[NewInstance] = nil
    end
end

local function optimizeInstance(instance)
    if instance:IsA("Texture") or instance:IsA("Decal") then
        instance:Destroy()
    elseif instance:IsA("ParticleEmitter") or instance:IsA("Trail") or instance:IsA("Smoke") or instance:IsA("Fire") or instance:IsA("Sparkles") then
        instance:Destroy()
    elseif instance:IsA("BasePart") and not instance:IsDescendantOf(game.Players.LocalPlayer.Character or workspace) then
        instance.Material = Enum.Material.SmoothPlastic
        instance.Reflectance = 0
    elseif instance:IsA("SurfaceAppearance") or instance:IsA("MeshPart") then
        if instance:IsA("MeshPart") then
            instance.Material = Enum.Material.SmoothPlastic
        end
    end
end

if _G.FastAttack then
    local _ENV = (getgenv or getrenv or getfenv)()

    local function SafeWaitForChild(parent, childName, timeout)
        local success, result = pcall(function()
            return parent:WaitForChild(childName, timeout or 5)
        end)
        return success and result or nil
    end

    local Players = game:GetService("Players")
    local ReplicatedStorage = game:GetService("ReplicatedStorage")
    local RunService = game:GetService("RunService")
    local Player = Players.LocalPlayer

    if not Player then return end

    local Modules = SafeWaitForChild(ReplicatedStorage, "Modules")
    if not Modules then return end

    local Net = SafeWaitForChild(Modules, "Net")
    if not Net then return end

    local RegisterAttack = SafeWaitForChild(Net, "RE/RegisterAttack")
    local RegisterHit = SafeWaitForChild(Net, "RE/RegisterHit")

    local Enemies = SafeWaitForChild(workspace, "Enemies")
    local Characters = SafeWaitForChild(workspace, "Characters")

    local Settings = {
        AutoClick = true,
        ClickDelay = 0.015,
        MaxTargets = 8
    }

    local FastAttack = _ENV.rz_FastAttack or {
        Distance = 55,
        attackMobs = true,
        attackPlayers = true
    }

    local function IsAlive(char)
        if not char then return false end
        local hum = char:FindFirstChildOfClass("Humanoid")
        return hum and hum.Health > 0
    end

    local function IsValidPlayer(char)
        if char == Player.Character or not IsAlive(char) then return false end
        
        if char:FindFirstChildOfClass("ForceField") then
            return false
        end

        return true
    end

    local function GetTargets()
        local targets = {}
        local primaryPart = nil
        
        local myChar = Player.Character
        if not IsAlive(myChar) then return nil, targets end
        
        local myRoot = myChar:FindFirstChild("HumanoidRootPart")
        if not myRoot then return nil, targets end

        if FastAttack.attackMobs and Enemies then
            for _, enemy in ipairs(Enemies:GetChildren()) do
                if #targets >= Settings.MaxTargets then break end
                if IsAlive(enemy) then
                    local part = enemy:FindFirstChild("Head") or enemy:FindFirstChild("HumanoidRootPart")
                    if part and (part.Position - myRoot.Position).Magnitude <= FastAttack.Distance then
                        if not primaryPart then primaryPart = part end
                        table.insert(targets, { enemy, part })
                    end
                end
            end
        end

        if FastAttack.attackPlayers and Characters then
            for _, char in ipairs(Characters:GetChildren()) do
                if #targets >= Settings.MaxTargets then break end
                
                if IsValidPlayer(char) then
                    local targetRoot = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Head")
                    if targetRoot and (targetRoot.Position - myRoot.Position).Magnitude <= FastAttack.Distance then
                        if not primaryPart then primaryPart = targetRoot end
                        table.insert(targets, { char, targetRoot })
                    end
                end
            end
        end

        return primaryPart, targets
    end

    function FastAttack:BladeHits()
        local myChar = Player.Character
        if not IsAlive(myChar) then return end

        local tool = myChar:FindFirstChildOfClass("Tool")
        if not tool or tool.ToolTip == "Gun" then return end

        local primaryPart, targets = GetTargets()
        if primaryPart and #targets > 0 then
            pcall(function()
                tool:Activate()
            end)

            RegisterAttack:FireServer(Settings.ClickDelay)
            RegisterHit:FireServer(primaryPart, targets)
        end
    end

    if not _ENV.FastAttackThread then
        _ENV.FastAttackThread = task.spawn(function()
            while true do
                if Settings.AutoClick then
                    pcall(function()
                        FastAttack:BladeHits()
                    end)
                end
                task.wait(Settings.ClickDelay)
            end
        end)
    end

    _ENV.rz_FastAttack = FastAttack
end

                textLabel.Name = "ESPText"
                textLabel.Size = UDim2.new(1, 0, 1, 0)
                textLabel.BackgroundTransparency = 1
                textLabel.TextScaled = false
                textLabel.TextSize = 14
                textLabel.Font = Enum.Font.SourceSansBold
                textLabel.TextStrokeTransparency = 0
                textLabel.Parent = bg
                
                bg.Parent = char
                
                local connection
                connection = RunService.Heartbeat:Connect(function()
                    if not ESPEnabled or not char or not char.Parent or humanoid.Health <= 0 then
                        bg:Destroy()
                        if connection then connection:Disconnect() end
                        return
                    end
                    
                    local teamColor = Color3.fromRGB(255, 255, 255)
                    local teamVal = player:FindFirstChild("Team") or (player.Team and player.Team.Name)
                    local teamStr = tostring(player.Team or "")
                    
                    if teamStr == "Pirates" or teamStr == "Pirate" then
                        teamColor = Color3.fromRGB(255, 50, 50)
                    elseif teamStr == "Marines" or teamStr == "Marine" then
                        teamColor = Color3.fromRGB(50, 100, 255)
                    end
                    
                    textLabel.TextColor3 = teamColor
                    
                    local level = "N/A"
                    local dataFolder = player:FindFirstChild("Data")
                    if dataFolder and dataFolder:FindFirstChild("Level") then
                        level = tostring(dataFolder.Level.Value)
                    elseif player:FindFirstChild("Level") then
                        level = tostring(player.Level.Value)
                    end
                    
                    local hp = math.floor(humanoid.Health)
                    local maxHp = math.floor(humanoid.MaxHealth)
                    
                    textLabel.Text = string.format("Lv. %s | %s | %d/%d HP", level, player.DisplayName, hp, maxHp)
                end)
            end
            
            if player.Character then
                createBillboard(player.Character)
            end
            player.CharacterAdded:Connect(createBillboard)
        end
        
        for _, p in ipairs(Players:GetPlayers()) do
            applyESP(p)
        end
        
        Players.PlayerAdded:Connect(applyESP)
    end,
